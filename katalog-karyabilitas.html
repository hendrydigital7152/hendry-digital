<script>

let jumlah = 1;
let harga = 100000;

function tampilkanTotal(){
    if(harga > 0){
        let total = jumlah * harga;
        document.getElementById("total").innerText =
            "Rp " + total.toLocaleString("id-ID");
    }
}

function tambah(){
    jumlah++;
    document.getElementById("jumlah").innerText = jumlah;
    tampilkanTotal();
}

function kurang(){
    if(jumlah > 1){
        jumlah--;
        document.getElementById("jumlah").innerText = jumlah;
        tampilkanTotal();
    }
}

function pesanWhatsApp(){
    let total = jumlah * harga;

    let pesan =
        "Halo, saya ingin memesan Kain Batik Ciprat%0A" +
        "Jumlah: " + jumlah + "%0A" +
        "Total: Rp " + total.toLocaleString("id-ID");

    window.open(
        "https://wa.me/628161949570?text=" + pesan,
        "_blank"
    );
}
